The problems go from the √n divisor loop to the full sieve, each adding one idea. Try each one first, and open the hint only when you are stuck.

1. Perfect Number

LeetCode 507 · Perfect Number

A number is perfect if it equals the sum of its divisors other than itself: 28 = 1 + 2 + 4 + 7 + 14. Read n and print YES or NO.

Limits: 1 ≤ n ≤ 10^8 · Idea: divisors in √n

Input
28
Output
YES
Input
7
Output
NO
Hint

Loop i up to √n and add both i and n / i. Don't count n itself, and add a perfect-square divisor only once.

Solution

Start the sum at 1 (1 divides every n > 1) and loop from 2 while i * i <= n, adding each pair. n = 1 has no proper divisors, so it is not perfect. O(√n) time.

#include <iostream>
using namespace std;

int main() {
    long long n;
    cin >> n;
    if (n <= 1) { cout << "NO\n"; return 0; }
    long long sum = 1;                       // 1 is a proper divisor of every n > 1
    for (long long i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i != n / i) sum += n / i;    // the pair, unless it is the same number
        }
    }
    cout << (sum == n ? "YES" : "NO") << "\n";
    return 0;
}

2. Four Divisors

LeetCode 1390 · Four Divisors

Read n numbers. For each number that has exactly four divisors, add up its divisors. Print the total (0 if there are none).

Limits: 1 ≤ n ≤ 10^4, 1 ≤ each number ≤ 10^5 · Idea: count and sum divisors in √n

Input
3
21 4 7
Output
32
Input
5
1 2 3 4 5
Output
0
Hint

Count the divisors with the √n loop, adding them up as you go. If the count is exactly 4, add their sum. (21 has 1, 3, 7, 21 → 32.)

Solution

For each number, the √n loop finds every divisor pair, counting and adding them. Stop early once the count passes 4. O(n · √m) time, where m is the largest number.

#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    long long total = 0;
    for (int k = 0; k < n; k++) {
        int x;
        cin >> x;
        int count = 0, sum = 0;
        for (int d = 1; d * d <= x && count <= 4; d++) {
            if (x % d == 0) {
                count++; sum += d;
                if (d != x / d) { count++; sum += x / d; }
            }
        }
        if (count == 4) total += sum;
    }
    cout << total << "\n";
    return 0;
}

3. Count Primes

LeetCode 204 · Count Primes

Read n. Print how many prime numbers are strictly less than n.

Limits: 0 ≤ n ≤ 5 × 10^6 · Idea: the basic sieve

Input
10
Output
4
Input
1
Output
0
Hint

This is exactly the sieve from class. The only difference is that the problem asks for primes strictly less than n.

Solution

Sieve up to n − 1, then count what is still marked. n below 3 has no primes under it. O(n log log n) time, O(n) space.

#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    if (n < 3) { cout << 0 << "\n"; return 0; }
    vector<bool> isPrime(n, true);             // indexes 0 .. n-1
    isPrime[0] = isPrime[1] = false;
    for (long long i = 2; i * i < n; i++)
        if (isPrime[i])
            for (long long j = i * i; j < n; j += i) isPrime[j] = false;
    int count = 0;
    for (int i = 2; i < n; i++) if (isPrime[i]) count++;
    cout << count << "\n";
    return 0;
}

4. Closest Primes in a Range

LeetCode 2523 · Closest Prime Numbers in Range

Read left and right. Among the primes in [left, right], print the two neighbours with the smallest gap (the first such pair if there is a tie). If there are fewer than two primes, print -1 -1.

Limits: 1 ≤ left ≤ right ≤ 10^6 · Idea: sieve once, then scan the range

Input
10 19
Output
11 13
Input
4 6
Output
-1 -1
Hint

Sieve up to right, then walk from left to right and compare each prime with the previous one.

Solution

After the sieve, remember the previous prime while scanning; each new prime makes a pair with it. Keep the pair with the smallest gap, replacing only on a strictly smaller gap so ties keep the first pair.

#include <iostream>
#include <vector>
using namespace std;

int main() {
    int left, right;
    cin >> left >> right;
    vector<bool> isPrime(right + 1, true);
    isPrime[0] = false;
    if (right >= 1) isPrime[1] = false;
    for (long long i = 2; i * i <= right; i++)
        if (isPrime[i])
            for (long long j = i * i; j <= right; j += i) isPrime[j] = false;
    int prev = -1, a = -1, b = -1;
    for (int x = left; x <= right; x++) {
        if (!isPrime[x]) continue;
        if (prev != -1 && (a == -1 || x - prev < b - a)) { a = prev; b = x; }
        prev = x;
    }
    cout << a << " " << b << "\n";
    return 0;
}

5. Prime Pairs With Target Sum

LeetCode 2761 · Prime Pairs With Target Sum

Read n. Find every pair of primes x ≤ y with x + y = n. Print the number of pairs, then each pair on its own line in increasing order of x.

Limits: 1 ≤ n ≤ 10^6 · Idea: sieve + O(1) lookups

Input
10
Output
2
3 7
5 5
Input
2
Output
0
Hint

Sieve up to n. For every x from 2 to n/2, check that isPrime[x] and isPrime[n - x] are both true.

Solution

The sieve turns each “is it prime?” into an O(1) lookup, so checking every x up to n/2 costs only O(n). Stopping at n/2 keeps x ≤ y and avoids listing a pair twice.

#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<bool> isPrime(n + 1, true);
    isPrime[0] = false;
    if (n >= 1) isPrime[1] = false;
    for (long long i = 2; i * i <= n; i++)
        if (isPrime[i])
            for (long long j = i * i; j <= n; j += i) isPrime[j] = false;
    vector<int> xs;
    for (int x = 2; x <= n / 2; x++)
        if (isPrime[x] && isPrime[n - x]) xs.push_back(x);
    cout << xs.size() << "\n";
    for (int x : xs) cout << x << " " << n - x << "\n";
    return 0;
}

6. Fast Factorisation (Smallest Prime Factor)

GeeksforGeeks · Prime Factorization using Sieve O(log n)

Read q, then q numbers. For each one, print its prime factors in increasing order (repeated as often as they divide it), on its own line.

Limits: 1 ≤ q ≤ 10^5, 2 ≤ each number ≤ 10^6 · Idea: a modified sieve

Input
3
12 97 360
Output
2 2 3
97
2 2 2 3 3 5
Hint

Use the same loops as the sieve. Instead of false, store spf[j] = i when spf[j] is not set yet. Then keep dividing x by spf[x].

Solution

spf[x] is the smallest prime factor of x. Building it costs one sieve; after that each query divides x by spf[x] until it reaches 1, which takes O(log x) steps, because every step at least halves x.

#include <iostream>
#include <vector>
using namespace std;

const int MAX = 1000000;

int main() {
    vector<int> spf(MAX + 1, 0);
    for (long long i = 2; i <= MAX; i++) {
        if (spf[i] != 0) continue;                 // already has a smaller prime factor
        spf[i] = i;                                // i is prime
        for (long long j = i * i; j <= MAX; j += i)
            if (spf[j] == 0) spf[j] = i;
    }
    int q;
    cin >> q;
    while (q--) {
        int x;
        cin >> x;
        bool first = true;
        while (x > 1) {
            if (!first) cout << " ";
            cout << spf[x];
            x /= spf[x];
            first = false;
        }
        cout << "\n";
    }
    return 0;
}

7. Counting Divisors

CSES 1713 · Counting Divisors

Read n, then n numbers. For each one, print how many divisors it has, on its own line.

Limits: 1 ≤ n ≤ 10^5, 1 ≤ each number ≤ 10^6 · Idea: precompute with the sieve's j += i loop

Input
3
16
17
18
Output
5
2
6
Hint

An O(√x) loop for each query is about 10^5 × 1000 = 10^8 operations, which is borderline. A faster way: for every i, add 1 to cnt[j] for each j = i, 2i, 3i, ….

Solution

Every i is a divisor of exactly its multiples, so the double loop gives every number its full divisor count. It runs MAX × (1 + 1/2 + 1/3 + …) ≈ MAX × ln MAX ≈ 1.4 × 10^7 steps, once; then each query is a lookup.

#include <iostream>
#include <vector>
using namespace std;

const int MAX = 1000000;

int main() {
    vector<int> cnt(MAX + 1, 0);
    for (int i = 1; i <= MAX; i++)
        for (int j = i; j <= MAX; j += i) cnt[j]++;
    int n;
    cin >> n;
    while (n--) {
        int x;
        cin >> x;
        cout << cnt[x] << "\n";
    }
    return 0;
}
