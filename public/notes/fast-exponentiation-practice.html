1. Pow(x, n)

LeetCode 50 · Pow(x, n)

Read a decimal number x and an integer n. Print xⁿ with exactly 5 decimal places. n can be negative.

Limits: −100 < x < 100, −2^31 ≤ n ≤ 2^31 − 1 · Idea: the squaring loop

Input
2.00000 10
Output
1024.00000
Input
2.10000 3
Output
9.26100
Input
2.00000 -2
Output
0.25000
Hint

x⁻ⁿ = 1 / xⁿ. And −2^31 can't be negated inside an int.

Solution

Run the loop on |n| (stored in a long long, so −2^31 can be negated), then take 1 / result for negative n. O(log |n|).

#include <bits/stdc++.h>
using namespace std;

int main() {
    double x;
    long long n;
    cin >> x >> n;
    long long e = n < 0 ? -n : n;
    double r = 1, b = x;
    while (e > 0) {
        if (e & 1) r *= b;
        b *= b;
        e >>= 1;
    }
    if (n < 0) r = 1 / r;
    cout << fixed << setprecision(5) << r << "\n";
    return 0;
}

2. Exponentiation

CSES 1095 · Exponentiation

Read n, then n lines with a and b. For each, print aᵇ mod 10⁹ + 7. Take 0⁰ = 1.

Limits: 1 ≤ n ≤ 2 × 10^5, 0 ≤ a, b ≤ 10^9 · Idea: modular fast power

Input
3
3 4
2 8
123 123
Output
81
256
921450052
Hint

2 × 10^5 queries with b up to 10^9: a plain loop would be 10^14 steps. Each query needs O(log b).

Solution

The modPow template for each query: about 30 steps each, so 6 × 10^6 in total.

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;

long long modPow(long long a, long long b, long long m) {
    long long r = 1 % m;
    a %= m;
    while (b > 0) {
        if (b & 1) r = r * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return r;
}

int main() {
    int n;
    cin >> n;
    while (n--) {
        long long a, b;
        cin >> a >> b;
        cout << modPow(a, b, MOD) << "\n";
    }
    return 0;
}

3. Count Good Numbers

LeetCode 1922 · Count Good Numbers

A digit string is good if the digits at even positions (0, 2, 4, …) are even and the digits at odd positions are prime (2, 3, 5 or 7). Read n and print how many good strings of length n there are, mod 10⁹ + 7. Leading zeros are allowed.

Limits: 1 ≤ n ≤ 10^15 · Idea: counting = a product of powers

Input
1
Output
5
Input
4
Output
400
Input
50
Output
564908303
Hint

How many choices does each even position have? Each odd position? How many even positions are there in a string of length n?

Solution

Even positions have 5 choices (0, 2, 4, 6, 8) and odd positions 4 (2, 3, 5, 7). There are ⌈n/2⌉ even positions and ⌊n/2⌋ odd ones, so the answer is 5^⌈n/2⌉ × 4^⌊n/2⌋ mod 10⁹ + 7. For n = 4: 25 × 16 = 400.

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;

long long modPow(long long a, long long b, long long m) {
    long long r = 1 % m;
    a %= m;
    while (b > 0) {
        if (b & 1) r = r * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return r;
}

int main() {
    long long n;
    cin >> n;
    long long evens = (n + 1) / 2, odds = n / 2;
    cout << modPow(5, evens, MOD) * modPow(4, odds, MOD) % MOD << "\n";
    return 0;
}

4. Super Pow

LeetCode 372 · Super Pow

Read a, then k, then the k digits of a huge exponent b (most significant first). Print aᵇ mod 1337.

Limits: 1 ≤ a ≤ 2^31 − 1, 1 ≤ k ≤ 2000 · Idea: the exponent is too big for any type

Input
2
2
1 0
Output
1024
Input
2147483647
3
2 0 0
Output
1198
Hint

Read b digit by digit, like building a number: if b = 10x + d, then aᵇ = (aˣ)¹⁰ × aᵈ.

Solution

Keep r = a^(digits read so far). Each new digit d turns r into r¹⁰ × aᵈ, all mod 1337. Every power is a small modPow, so it is O(k) steps.

#include <bits/stdc++.h>
using namespace std;

long long modPow(long long a, long long b, long long m) {
    long long r = 1 % m;
    a %= m;
    while (b > 0) {
        if (b & 1) r = r * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return r;
}

int main() {
    long long a;
    int k;
    cin >> a >> k;
    long long r = 1;
    for (int i = 0; i < k; i++) {
        int d;
        cin >> d;
        r = modPow(r, 10, 1337) * modPow(a, d, 1337) % 1337;   // b = 10x + d
    }
    cout << r << "\n";
    return 0;
}

5. Exponentiation II

CSES 1712 · Exponentiation II

Read n, then n lines with a, b and c. For each, print a^(b^c) mod 10⁹ + 7.

Limits: 1 ≤ n ≤ 10^5, 0 ≤ a, b, c ≤ 10^9 · Idea: Fermat reduces the exponent

Input
3
3 7 1
15 2 2
3 4 5
Output
2187
50625
763327764
Hint

b^c is far too big to compute. For a prime p and a not divisible by p, a^e ≡ a^(e mod (p − 1)). So which modulus do you use for b^c?

Solution

Compute e = b^c mod (p − 1) with fast power, then a^e mod p. Fermat needs a not divisible by p; here a ≤ 10^9 < p, so that only fails for a = 0, where the answer is 1 if b^c = 0 (b = 0 and c > 0) and 0 otherwise.

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;

long long modPow(long long a, long long b, long long m) {
    long long r = 1 % m;
    a %= m;
    while (b > 0) {
        if (b & 1) r = r * a % m;
        a = a * a % m;
        b >>= 1;
    }
    return r;
}

int main() {
    int n;
    cin >> n;
    while (n--) {
        long long a, b, c;
        cin >> a >> b >> c;
        if (a == 0) { cout << (b == 0 && c > 0 ? 1 : 0) << "\n"; continue; }
        long long e = modPow(b, c, MOD - 1);      // b^c mod (p - 1)
        cout << modPow(a, e, MOD) << "\n";
    }
    return 0;
}

6. Fibonacci Numbers

CSES 1722 · Fibonacci Numbers

F(0) = 0, F(1) = 1, and F(n) = F(n − 1) + F(n − 2). Read n and print F(n) mod 10⁹ + 7.

Limits: 0 ≤ n ≤ 10^18 · Idea: matrix power

Input
10
Output
55
Input
0
Output
0
Hint

A loop to 10^18 is far too slow. The matrix [[1, 1], [1, 0]] raised to the power n holds F(n). Apply fast power to matrices.

Solution

Same loop as modPow, but the “numbers” are 2 × 2 matrices and the multiplication is matrix multiplication. Mⁿ = [[F(n+1), F(n)], [F(n), F(n−1)]], so the answer is the top-right entry. O(log n) matrix products.

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;
typedef array<array<long long, 2>, 2> Mat;

Mat mul(const Mat &A, const Mat &B) {
    Mat C = {{{0, 0}, {0, 0}}};
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++)
                C[i][j] = (C[i][j] + A[i][k] * B[k][j]) % MOD;
    return C;
}

int main() {
    long long n;
    cin >> n;
    Mat r = {{{1, 0}, {0, 1}}}, a = {{{1, 1}, {1, 0}}};   // r starts as the identity
    while (n > 0) {
        if (n & 1) r = mul(r, a);
        a = mul(a, a);
        n >>= 1;
    }
    cout << r[0][1] << "\n";
    return 0;
}
